мясо · ${fr.label}, ${formatTime(fr.remaining)}`;
  }

function craftRecipe(r,qty=1){
  if(!canCraftAt(r))return toast('Для этого нужен костёр в лагере.','bad');
  if(r.unique&&(hero.inv[r.id]||0)>0)return toast('Этот предмет уже изготовлен.','bad');
  if(r.id==='bone_tip'&&((hero.inv.bone_tip||0)>0||(hero.inv.improved_spear||0)>0))return toast('Нужный наконечник уже готов или использован. Новые применения появятся позже.','bad');
  const max=maxCraftable(r);qty=Math.max(1,Math.min(max,Math.floor(Number(qty)||1)));if(r.unique)qty=1;
  if(max<=0)return toast(`Не хватает материалов: ${missingMaterials(r).join(' · ')}`,'bad');
  const result=RULES.applyRecipe(hero.inv,r,qty);if(!result.ok)return toast('Не хватает материалов.','bad');hero.inv=result.inventory;
  if(r.id==='roasted_root')addKnowledgePlant('root','cooked');
  if(r.id==='improved_spear'){
    hero.weaponQuality.improved_spear=clamp(1.04+(hero.skills.crafting.mastery||0)*.0015,1.04,1.22);
    hero.equipment.weapon='improved_spear';hero.hotbar=hero.hotbar.map(id=>id==='spear'?'improved_spear':id);
  }
  gainSkill('crafting',0.22*Math.sqrt(qty));if(r.id==='cooked_meat'||r.id==='roasted_root')gainSkill('survival',0.12*Math.sqrt(qty));
  toast(`Создано: ${r.name} ×${qty}`);renderHotbar();renderPanels();saveGame(true);
}
function experiment(a,b,method){
  if(!a||!b||a===b)return toast('Нужно выбрать два разных материала.','bad');
  if((hero.inv[a]||0)<1||(hero.inv[b]||0)<1)return toast('Материалов нет в сумке.','bad');
  removeInv(a);removeInv(b);const methodName=EXPERIMENT_METHODS[method]||method;
  const rx=HIDDEN_REACTIONS.find(r=>((r.a===a&&r.b===b)||(r.a===b&&r.b===a))&&r.method===method);
  if(rx){
    addInv(rx.out,1);if(!hero.knowledge.reactions.includes(rx.label))hero.knowledge.reactions.push(rx.label);
    rx.facts.forEach(([id,f])=>addKnowledgePlant(id,f));gainSkill('research',0.9);toast(`✨ Открыта реакция: ${rx.label}`);
  }else{
    gainSkill('research',0.18);const m=hero.skills.research.mastery;
    const observation=m>=18?'Вещества заметно изменились, но выбранный способ не завершил полезную реакцию.':m>=7?'Один материал изменил запах или структуру другого, но практического результата нет.':'Подтверждённого полезного эффекта не обнаружено.';
    const note=`${ITEMS[a].name} + ${ITEMS[b].name}; ${methodName.toLowerCase()}: ${observation}`;
    if(!hero.knowledge.experimentNotes.includes(note))hero.knowledge.experimentNotes.unshift(note);hero.knowledge.experimentNotes=hero.knowledge.experimentNotes.slice(0,40);
    toast(`Материалы испорчены. ${observation}`,'bad');
  }
  renderPanels();saveGame(true);
}
