function assimilateStats(dt){
  if(hero.hunger<15||totalLatent()<=.0001)return;
  let budget=dt*(.018+.0025*hero.stats.endurance/10);
  for(let guard=0;guard<8&&budget>0;guard++){
    const ids=Object.keys(STAT_META).filter(id=>hero.statLatent[id]>.00001);
    if(!ids.length)break;

    const take=Math.min(budget,hero.statLatent[id]);

}
