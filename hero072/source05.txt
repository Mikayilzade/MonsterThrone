function update(dt){
  if(world.paused||hero.respawning)return;
  world.time+=dt;hero.playTime+=dt;autoSaveTimer+=dt;attackCd=Math.max(0,attackCd-dt);dodgeCd=Math.max(0,dodgeCd-dt);hero.repellent=Math.max(0,(hero.repellent||0)-dt);ensureHeroFields();for(const sk of Object.values(hero.skills))sk.readiness=Math.max(20,sk.readiness-dt*.018);
  if(autoSaveTimer>12){autoSaveTimer=0;saveGame(true,'auto');}
  const hungerDrain=.055/skillFactor('survival',.0025,.0004);hero.hunger=Math.max(0,hero.hunger-dt*hungerDrain);if(hero.hunger<=0){hero.hp-=dt*2.2;if(hero.hp<=0)die('голод');}
  hero.stamina=clamp(hero.stamina+dt*(10+hero.stats.endurance*.2),0,hero.maxStamina);if(hero.tempPowerTime>0){hero.tempPowerTime-=dt;if(hero.tempPowerTime<=0){hero.tempPower=0;toast('Временная сила рассеялась.');}}
  assimilateStats(dt);
  if(hero.level<hero.ceiling&&hero.level<999){hero.levelProgress=(hero.levelProgress||0)+dt/60*growthSpeed();while(hero.levelProgress>=1&&hero.level<hero.ceiling){hero.level++;hero.levelProgress-=1;toast(`Уровень вырос до ${hero.level}.`);}}
  else hero.levelProgress=Math.min(hero.levelProgress||0,.999);
  let v=inputVec();const terrain=terrainAt(hero.x,hero.y);let speed=hero.move*(terrain==='swamp'?.72:terrain==='rock'?.83:1);
  if(dodgeTime>0){dodgeTime-=dt;moveHero(dodgeVX*dt,dodgeVY*dt);} else moveHero(v.x*speed*dt,v.y*speed*dt);refreshSectorWorld();
  updateEnemies(dt);updateTraps();updateCarcasses(dt);updateHud();updateQuest();
}

function updateEnemies(dt){
  const sanitized=SECTORS.partitionLandEntities(world.enemies,sectorConfig);for(const e of sanitized.removed){if(target===e)target=null;if(e.uid)sectorManager.remove(e.uid);}world.enemies=sanitized.valid;
  for(const e of world.enemies){if(e.dead)continue;const ox=e.x,oy=e.y,d=ENEMY_TYPES[e.type];e.attackCd=Math.max(0,e.attackCd-dt);e.slow=Math.max(0,e.slow-dt);e.think-=dt;let toHero=dist(e,hero),speed=d.speed*(e.slow>0?.42:1);if(hero.repellent>0&&e.type==='tick'&&toHero<190){const a=Math.atan2(e.y-hero.y,e.x-hero.x);e.x+=Math.cos(a)*speed*1.4*dt;e.y+=Math.sin(a)*speed*1.4*dt;if(!SECTORS.isLand(e.x,e.y,sectorConfig)){e.x=ox;e.y=oy;}continue;}
    if((d.role==='predator'||d.role==='territorial'||d.role==='scavenger')&&toHero<d.aggro){e.state='attack';const a=Math.atan2(hero.y-e.y,hero.x-e.x);e.x+=Math.cos(a)*speed*dt;e.y+=Math.sin(a)*speed*dt;if(toHero<d.radius+hero.radius+8&&e.attackCd<=0){e.attackCd=1.0;heroHit(e);}}
    else if(d.role==='grazer'&&toHero<90){e.state='flee';const a=Math.atan2(e.y-hero.y,e.x-hero.x);e.x+=Math.cos(a)*speed*1.15*dt;e.y+=Math.sin(a)*speed*1.15*dt;}
    else{e.state='wander';if(e.think<=0){e.think=rnd(4,1.2);e.dir+=rnd(2,-2);}e.x+=Math.cos(e.dir)*speed*.35*dt;e.y+=Math.sin(e.dir)*speed*.35*dt;}
    if(!Number.isFinite(e.x)||!Number.isFinite(e.y)||!SECTORS.isLand(e.x,e.y,sectorConfig)){e.x=ox;e.y=oy;e.dir+=Math.PI;}
    if(d.role==='predator'){const prey=nearest(world.enemies.filter(x=>!x.dead&&x!==e&&ENEMY_TYPES[x.type].role==='grazer'),e,100);if(prey&&Math.random()<dt*.7){prey.hp-=d.attack*.5;if(prey.hp<=0)killEnemy(prey,'monster');}}
  }}
function updateTraps(){
  for(const o of world.objects){
    if(o.kind!=='trap'||!o.armed)continue;
    const e=nearest(world.enemies.filter(x=>!x.dead),o,18);
    if(e){
      o.armed=false;const quality=o.quality||1,dmg=Math.round((o.poisoned?78:34)*quality*skillFactor('trapping',.004,.0007));e.hp-=dmg;e.slow=4*clamp(quality,.8,2.2);gainSkill('trapping',0.35);
      floatDamage(e.x,e.y,'-'+dmg,'out');combat(`Ловушка → ${ENEMY_TYPES[e.type].name}: ${dmg}`,'out');
      if(e.hp<=0)killEnemy(e,'hero');
      setTimeout(()=>{const i=world.objects.indexOf(o);if(i>=0)world.objects.splice(i,1);},300);
    }
  }
}

      if(selectedCarcass===c)selectedCarcass=null;
      world.carcasses.splice(i,1);
    }
  }
}
